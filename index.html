async function executeGitHubDeploy() { const token = document.getElementById('github-token').value.trim(); // Pega o nome e já formata no padrão correto do GitHub (sem espaços, minúsculo) let repoNameRaw = document.getElementById('github-repo-name').value.trim(); let repoName = repoNameRaw.toLowerCase().normalize("NFD").replace(/[\u0300-\u036f]/g, "").replace(/\s+/g, '-').replace(/[^a-z0-9-_]/g, ''); const btn = document.getElementById('btn-github-deploy'); if(!token || !repoName) return showToast('Preencha o nome do App e o seu Token.', 'error'); if(!token.startsWith('ghp_')) return showToast('O token deve começar com "ghp_"', 'error'); localStorage.setItem('fl_github_token', token); btn.innerHTML = ' Criando repositório...'; try { // 1. Criar Repositório const createRes = await fetch('https://api.github.com/user/repos', { method: 'POST', headers: { 'Authorization': `Bearer ${token}`, 'Content-Type': 'application/json' }, body: JSON.stringify({ name: repoName, description: 'Aplicativo gerado via Fábrica Livre de forma autônoma.', private: false, auto_init: true }) }); if(!createRes.ok) { const errInfo = await createRes.json(); if(errInfo.message.includes('name already exists')) throw new Error('Já existe um repositório com esse nome no seu GitHub.'); if(createRes.status === 401) throw new Error('Token inválido. Verifique se copiou certo.'); throw new Error('Erro GitHub: ' + errInfo.message); // Agora mostra o erro REAL } const repoData = await createRes.json(); const owner = repoData.owner.login; // 2. Fazer Push do Código btn.innerHTML = ' Injetando código...'; const contentEncoded = btoa(unescape(encodeURIComponent(currentCodeToExport))); const uploadRes = await fetch(`https://api.github.com/repos/${owner}/${repoName}/contents/index.html`, { method: 'PUT', headers: { 'Authorization': `Bearer ${token}`, 'Content-Type': 'application/json' }, body: JSON.stringify({ message: '🚀 Deploy autônomo via Fábrica Livre', content: contentEncoded }) }); if(!uploadRes.ok) { const uploadErr = await uploadRes.json(); throw new Error('Erro ao enviar arquivo: ' + uploadErr.message); } showToast('✅ Deploy concluído com sucesso!', 'success'); window.open(repoData.html_url, '_blank'); closeModal('modal-github'); } catch(e) { showToast(e.message, 'error'); } finally { btn.innerHTML = ' Enviar Código'; document.getElementById('github-repo-name').value = repoName; // Atualiza pro nome corrigido na tela } }
